import net from "node:net";
import type { ExtensionAPI } from "@earendil-works/pi-coding-agent";
import { Type } from "typebox";

type StacksResponse = {
  ok: boolean;
  message: string;
};

type CardRequest = {
  action: "updateLocalCard" | "finishLocalCardRefinement" | "startLocalCardWork";
  cardId: string;
  title?: string;
  content?: string;
};

export default function stacksCards(pi: ExtensionAPI) {
  const cardId = process.env.STACKS_CARD_ID;
  const thread = process.env.STACKS_CARD_THREAD;
  const socketPath = process.env.STACKS_AUTOMATION_SOCKET;

  // Ordinary terminal Pi sessions should not know about or mutate Stacks cards.
  if (!cardId || !socketPath || !cardId.startsWith("local:") || thread !== "planning") return;

  const request = (payload: Omit<CardRequest, "cardId">, signal?: AbortSignal) =>
    sendRequest(socketPath, { ...payload, cardId }, signal);

  pi.registerTool({
    name: "update_card_description",
    label: "Update card description",
    description: "Replace the current Stacks local card description with the complete updated description. This is durable card state, not a chat draft. The active card is scoped by Stacks and cannot be selected by the model.",
    promptSnippet: "Persist an approved description to the active local Stacks card",
    promptGuidelines: [
      "Use update_card_description only after the user asks to save or update the active card description; send the complete replacement description, not a patch or commentary.",
      "Do not claim a card description was saved unless update_card_description or finish_refinement succeeds.",
    ],
    parameters: Type.Object({
      description: Type.String({ description: "The complete Markdown description that should replace the card's current description", minLength: 1 }),
    }),
    async execute(_toolCallId, params, signal) {
      const response = await request({ action: "updateLocalCard", content: params.description }, signal);
      return {
        content: [{ type: "text", text: response.message }],
        details: { cardId, updated: "description" },
      };
    },
  });

  pi.registerTool({
    name: "update_card_title",
    label: "Update card title",
    description: "Replace the title of the active Stacks local card. The active card is scoped by Stacks and cannot be selected by the model.",
    promptSnippet: "Persist an approved title to the active local Stacks card",
    promptGuidelines: [
      "Use update_card_title only after the user asks to save or update the active card title.",
    ],
    parameters: Type.Object({
      title: Type.String({ description: "The complete replacement card title", minLength: 1 }),
    }),
    async execute(_toolCallId, params, signal) {
      const response = await request({ action: "updateLocalCard", title: params.title }, signal);
      return {
        content: [{ type: "text", text: response.message }],
        details: { cardId, updated: "title" },
      };
    },
  });

  pi.registerTool({
    name: "start_work",
    label: "Start card work",
    description: "Ask Stacks to start work on the active local card. Stacks validates workflow state and owns branch, sibling worktree, environment, persistence, and Work Pi session setup. The operation is idempotent when work is already started.",
    promptSnippet: "Start the active Ready-for-agent local card through Stacks",
    promptGuidelines: [
      "Use start_work only when the user explicitly asks to start agent work on the active card.",
      "Never create the card branch or worktree directly when start_work is available; Stacks must own that lifecycle.",
    ],
    parameters: Type.Object({}),
    async execute(_toolCallId, _params, signal) {
      const response = await request({ action: "startLocalCardWork" }, signal);
      return {
        content: [{ type: "text", text: response.message }],
        details: { cardId, started: true },
      };
    },
  });

  pi.registerTool({
    name: "finish_refinement",
    label: "Finish refinement",
    description: "Atomically save the complete final brief to the active Stacks local card and move it to Ready for agent. Call this only after explicit user approval.",
    promptSnippet: "Save an approved final brief and move the active local card to Ready for agent",
    promptGuidelines: [
      "Use finish_refinement only when the user explicitly approves the final brief or explicitly asks to finish refinement.",
      "The finish_refinement description must be a self-contained implementation brief with outcome, acceptance criteria, technical approach, risks or open questions, and validation plan where applicable.",
    ],
    parameters: Type.Object({
      description: Type.String({ description: "The complete final Markdown implementation brief", minLength: 1 }),
      title: Type.Optional(Type.String({ description: "An approved replacement title, when the title also changed", minLength: 1 })),
    }),
    async execute(_toolCallId, params, signal) {
      const response = await request({
        action: "finishLocalCardRefinement",
        content: params.description,
        title: params.title,
      }, signal);
      return {
        content: [{ type: "text", text: response.message }],
        details: { cardId, updated: params.title ? ["title", "description", "status"] : ["description", "status"] },
      };
    },
  });
}

function sendRequest(socketPath: string, request: CardRequest, signal?: AbortSignal): Promise<StacksResponse> {
  return new Promise((resolve, reject) => {
    let settled = false;
    let response = "";
    const socket = net.createConnection(socketPath);

    const finish = (error?: Error, value?: StacksResponse) => {
      if (settled) return;
      settled = true;
      signal?.removeEventListener("abort", abort);
      socket.destroy();
      if (error) reject(error);
      else resolve(value!);
    };
    const abort = () => finish(new Error("Stacks card update was cancelled"));

    signal?.addEventListener("abort", abort, { once: true });
    if (signal?.aborted) return abort();

    const timeoutMs = request.action === "startLocalCardWork" ? 12 * 60_000 : 10_000;
    socket.setTimeout(timeoutMs, () => finish(new Error("Stacks did not respond to the card request")));
    socket.on("connect", () => socket.write(`${JSON.stringify(request)}\n`));
    socket.on("data", (chunk) => {
      response += chunk.toString("utf8");
      const newline = response.indexOf("\n");
      if (newline < 0) return;
      try {
        const parsed = JSON.parse(response.slice(0, newline)) as StacksResponse;
        if (!parsed.ok) finish(new Error(parsed.message || "Stacks rejected the card update"));
        else finish(undefined, parsed);
      } catch (error) {
        finish(error instanceof Error ? error : new Error(String(error)));
      }
    });
    socket.on("error", (error) => finish(new Error(`Could not contact Stacks: ${error.message}`)));
    socket.on("end", () => {
      if (!settled) finish(new Error("Stacks closed the card update connection without responding"));
    });
  });
}
